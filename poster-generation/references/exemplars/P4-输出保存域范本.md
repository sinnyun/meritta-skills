### P4-05 图片导出功能范本（必选）

**任务目标**：在生成的 HTML 海报中嵌入「导出为图片」功能，用户点击按钮即可一键导出高清 PNG，颜色和字体与页面完全一致

**执行步骤**：

1. **引入 html-to-image 库**
   - 在 `</body>` 前引入 html-to-image CDN 脚本
   - 使用稳定版本号，确保兼容性

2. **添加导出按钮**
   - 在海报容器外部添加固定定位的导出按钮
   - 按钮样式与海报风格协调

3. **编写导出逻辑**
   - 监听按钮点击事件
   - 调用 html-to-image 的 `toPng` 方法截取 `.poster` 元素
   - 自动触发 PNG 下载

**为什么用 html-to-image 而不是 html2canvas**：

html2canvas 在 canvas 2D 上下文中重新实现 CSS 渲染，与浏览器原生排版引擎存在差异，导致：
- 中文字体字距、字重渲染偏差
- 渐变色、阴影颜色不一致
- `background-clip: text`、`backdrop-filter` 等属性不支持

html-to-image 使用 SVG foreignObject 序列化 DOM，由**浏览器原生渲染引擎**绘制到 canvas，导出结果与页面显示完全一致，无需任何 CSS 兼容性 hack。

**输出示例**：

在 HTML 的 `</body>` 标签前插入以下代码：

```html
<!-- ========== 图片导出功能 ========== -->
<!-- html-to-image 库：SVG foreignObject 方案，导出与页面完全一致 -->
<!-- 多 CDN 源逐级回退，确保内网/外网均可用 -->
<script>
  (function() {
    var cdnSources = [
      'https://unpkg.com/html-to-image@1.11.13/dist/html-to-image.js',
      'https://cdn.jsdelivr.net/npm/html-to-image@1.11.13/dist/html-to-image.js',
      'https://fastly.jsdelivr.net/npm/html-to-image@1.11.13/dist/html-to-image.js',
      'https://registry.npmmirror.com/html-to-image/1.11.13/files/dist/html-to-image.js'
    ];
    var idx = 0;
    function tryLoad() {
      if (idx >= cdnSources.length) {
        console.error('[导出] 所有 CDN 源均不可达，导出功能不可用');
        var btn = document.querySelector('.export-btn');
        if (btn) { btn.disabled = true; btn.querySelector('.text').textContent = '导出不可用（网络问题）'; }
        return;
      }
      var s = document.createElement('script');
      s.src = cdnSources[idx];
      s.onload = function() { console.log('[导出] html-to-image 加载成功:', cdnSources[idx]); };
      s.onerror = function() { console.warn('[导出] CDN 不可达，尝试下一个:', cdnSources[idx]); idx++; tryLoad(); };
      document.head.appendChild(s);
    }
    tryLoad();
  })();
</script>

<!-- 导出按钮样式 -->
<style>
  .export-btn-wrapper {
    position: fixed;
    bottom: 24px;
    right: 24px;
    z-index: 9999;
  }
  .export-btn {
    display: flex;
    align-items: center;
    gap: 8px;
    padding: 12px 24px;
    background: linear-gradient(135deg, #00f0ff, #b026ff);
    color: #fff;
    border: none;
    border-radius: 8px;
    font-size: 14px;
    font-weight: 600;
    cursor: pointer;
    box-shadow: 0 4px 20px rgba(0, 240, 255, 0.4);
    transition: all 0.3s;
    font-family: "Microsoft YaHei", "PingFang SC", sans-serif;
  }
  .export-btn:hover {
    transform: translateY(-2px);
    box-shadow: 0 6px 28px rgba(0, 240, 255, 0.6);
  }
  .export-btn:disabled {
    opacity: 0.6;
    cursor: wait;
  }
  .export-btn .icon {
    font-size: 18px;
    line-height: 1;
  }
</style>

<!-- 导出按钮 -->
<div class="export-btn-wrapper">
  <button class="export-btn" onclick="exportPoster()">
    <span class="icon">📷</span>
    <span class="text">导出为图片</span>
  </button>
</div>

<!-- 导出逻辑脚本 -->
<script>
  function exportPoster() {
    var btn = document.querySelector('.export-btn');
    var btnWrapper = document.querySelector('.export-btn-wrapper');
    var originalText = btn.querySelector('.text').textContent;
    btn.disabled = true;
    btn.querySelector('.text').textContent = '正在导出...';

    // 检查 html-to-image 是否加载成功
    if (typeof htmlToImage === 'undefined') {
      alert('导出库未加载，请检查网络连接后刷新页面重试');
      btn.disabled = false;
      btn.querySelector('.text').textContent = originalText;
      return;
    }

    // 临时隐藏导出按钮，避免出现在截图中
    btnWrapper.style.visibility = 'hidden';

    var posterEl = document.querySelector('.poster');
    if (!posterEl) {
      alert('未找到海报元素（.poster）');
      btnWrapper.style.visibility = '';
      btn.disabled = false;
      btn.querySelector('.text').textContent = originalText;
      return;
    }

    // 等待字体加载完成后再导出，避免字体回退导致失真
    var fontReady = document.fonts && document.fonts.ready ? document.fonts.ready : Promise.resolve();

    fontReady.then(function() {
      // 额外延迟 200ms 确保所有渲染完成
      return new Promise(function(resolve) { setTimeout(resolve, 200); });
    }).then(function() {
      // 固定使用 3 倍分辨率，确保高清且一致
      var pixelRatio = 3;

      return htmlToImage.toPng(posterEl, {
        pixelRatio: pixelRatio,
        cacheBust: true,
        backgroundColor: '#ffffff',
        filter: function(node) {
          // 过滤掉导出按钮自身（双重保险）
          return !node.classList || !node.classList.contains('export-btn-wrapper');
        }
      });
    }).then(function(dataUrl) {
      // 从 dataUrl 转为 Blob 下载
      fetch(dataUrl)
        .then(function(res) { return res.blob(); })
        .then(function(blob) {
          var url = URL.createObjectURL(blob);
          var a = document.createElement('a');
          a.href = url;
          // 文件名取 HTML 文件名，扩展名改为 png
          var fileName = '海报.png';
          try {
            var path = window.location.pathname;
            var base = path.substring(path.lastIndexOf('/') + 1);
            if (base && base.endsWith('.html')) {
              fileName = base.replace('.html', '.png');
            }
          } catch(e) {}
          a.download = fileName;
          document.body.appendChild(a);
          a.click();
          document.body.removeChild(a);
          URL.revokeObjectURL(url);
        });
    }).catch(function(err) {
      console.error('导出失败:', err);
      alert('导出失败，请重试');
    }).finally(function() {
      // 无论成功失败，最后恢复按钮
      btnWrapper.style.visibility = '';
      btn.disabled = false;
      btn.querySelector('.text').textContent = originalText;
    });
  }
</script>
<!-- ========== 图片导出功能结束 ========== -->
```

**注意事项**：

1. **CDN 多源回退**：按顺序尝试 4 个 CDN 源加载 `html-to-image@1.11.13`：① unpkg.com ② cdn.jsdelivr.net ③ fastly.jsdelivr.net ④ registry.npmmirror.com（国内镜像），全部不可达时禁用导出按钮并提示
2. **按钮位置**：固定在右下角，z-index 设为 9999 确保始终可见
3. **截图前隐藏按钮**：导出时用 `visibility: hidden` 临时隐藏按钮（比 `display: none` 更安全，不影响布局），导出完成后恢复
4. **导出分辨率**：固定使用 `pixelRatio: 3`（3 倍超采样），确保所有设备导出一致且高清
5. **文件命名**：自动取 HTML 文件名作为 PNG 文件名
6. **按钮样式**：使用渐变背景和阴影，与科技风海报协调；可根据海报风格调整配色
7. **兼容性**：html-to-image 兼容所有现代浏览器（Chrome/Firefox/Safari/Edge）
8. **无需 CSS 兼容 hack**：html-to-image 基于 SVG foreignObject，使用浏览器原生渲染引擎，天然支持 `background-clip: text`、`backdrop-filter`、`filter`、`mask`、`clip-path` 等现代 CSS 属性，导出颜色/字体与页面完全一致
9. **按钮恢复时机**：使用 `.finally()` 确保无论成功或失败，按钮状态都在最后恢复，避免重复点击
10. **跨域图片**：如果海报中包含跨域 `<img>`，需确保图片服务器返回 CORS 头，否则 html-to-image 无法序列化该图片
11. **字体加载等待**：导出前通过 `document.fonts.ready` 等待 Web 字体加载完成，再延迟 200ms 确保渲染完毕，避免字体回退导致文字失真
12. **CDN 回退**：主 CDN 不可达时自动尝试 unpkg 备用源；如果两个 CDN 都不可达，提示用户检查网络
13. **固定分辨率**：使用固定 `pixelRatio: 3` 而非动态计算，确保不同设备导出结果一致
